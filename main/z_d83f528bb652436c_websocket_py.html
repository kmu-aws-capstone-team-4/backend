48class BaseWebSocketConsumer(AsyncWebsocketConsumer): 

49 """프로젝트 공통 WebSocket consumer 베이스. 

50 

51 서브클래스는 handle_connect / handle_disconnect / handle_message 를 오버라이드한다. 

52 JSON 직렬화/역직렬화와 에러 처리는 베이스에서 담당한다. 

53 """ 

54 

55 async def connect(self) -> None: 

56 await self.accept() 

57 logger.info("ws_connected", channel=self.channel_name) 

58 await self.handle_connect() 

59 

60 async def disconnect(self, code: int) -> None: 

61 logger.info("ws_disconnected", channel=self.channel_name, code=code) 

62 await self.handle_disconnect(code) 

63 

64 async def receive(self, text_data: str = "", bytes_data: bytes = b"") -> None: 

65 try: 

66 data = json.loads(text_data) if text_data else {} 

67 except json.JSONDecodeError: 

68 await self.reply({"error": "invalid JSON"}) 

69 return 

70 await self.handle_message(data) 

71 

72 async def reply(self, data: dict) -> None: 

73 """연결된 클라이언트에 JSON 메시지를 전송한다.""" 

74 await self.send(text_data=json.dumps(data, ensure_ascii=False)) 

75 

76 # ── 서브클래스 훅 ──────────────────────────────────────────────────────── 

77 

78 async def handle_connect(self) -> None: 

79 """연결 수립 후 호출된다. 그룹 참가 등 초기화 로직을 여기에 작성한다.""" 

80 

81 async def handle_disconnect(self, code: int) -> None: 

82 """연결 종료 시 호출된다. 그룹 탈퇴 등 정리 로직을 여기에 작성한다.""" 

83 

84 async def handle_message(self, data: dict) -> None: 

85 """클라이언트 메시지 수신 시 호출된다. 서브클래스에서 필요 시 구현한다."""

88class UserWebSocketConsumer(BaseWebSocketConsumer): 

89 """인증된 특정 사용자 전용 WebSocket consumer 베이스. 

90 

91 연결 시 JWT 토큰을 검증하고, 해당 사용자의 개인 그룹(user_{id})에 자동 참가한다. 

92 인증 실패 시 연결을 즉시 닫는다. 

93 

94 채널 레이어를 통해 외부(Celery task 등)에서 특정 사용자에게 메시지를 push할 수 있다. 

95 

96 그룹 이름 규칙: "user_{user_id}" 

97 메시지 type 규칙: "user.message" (Channels는 type의 '.'을 '_'로 변환해 메서드를 호출한다) 

98 """ 

99 

100 _user_group: str = "" 

101 

102 async def connect(self) -> None: 

103 user = await self._authenticate() 

104 if user is None: 

105 await self.close(code=4001) 

106 return 

107 

108 self.user = user 

109 self._user_group = f"user_{user.pk}" 

110 

111 await self.accept() 

112 await self.channel_layer.group_add(self._user_group, self.channel_name) 

113 logger.info("ws_user_connected", channel=self.channel_name, user_id=user.pk) 

114 await self.handle_connect() 

115 

116 async def disconnect(self, code: int) -> None: 

117 if self._user_group: 

118 await self.channel_layer.group_discard(self._user_group, self.channel_name) 

119 logger.info("ws_user_disconnected", channel=self.channel_name, code=code) 

120 await self.handle_disconnect(code) 

121 

122 # ── 채널 레이어 메시지 핸들러 ──────────────────────────────────────────── 

123 # group_send({"type": "user.message", "payload": {...}}) 호출 시 실행된다. 

124 

125 async def user_message(self, event: dict) -> None: 

126 """채널 레이어에서 수신한 메시지를 클라이언트로 전달한다.""" 

127 await self.reply(event.get("payload", {})) 

128 

129 # ── 사용자 재조회 ───────────────────────────────────────────────────────── 

130 

131 async def get_fresh_user(self): 

132 """DB에서 사용자를 재조회해 최신 상태를 반환한다. 

133 

134 self.user 는 연결 시점의 스냅샷이므로, 권한 체크나 중요한 데이터 처리 시 

135 이 메서드로 최신 상태를 확인한다. 계정이 비활성화됐으면 None 을 반환한다. 

136 

137 사용 예:: 

138 

139 async def handle_message(self, data: dict) -> None: 

140 user = await self.get_fresh_user() 

141 if user is None: 

142 await self.close(code=4001) 

143 return 

144 # 이후 user 로 처리 

145 """ 

146 from django.contrib.auth import get_user_model 

147 User = get_user_model() 

148 try: 

149 return await User.objects.aget(pk=self.user.pk, is_active=True) 

150 except User.DoesNotExist: 

151 return None

155 async def _authenticate(self): 

156 """쿼리스트링의 단기 티켓을 검증하고 User 인스턴스를 반환한다. 

157 

158 검증 실패 시 None 을 반환한다. 

159 티켓은 1회용이며 검증 즉시 삭제된다 (60초 TTL). 

160 

161 URL 예시: ws://host/ws/notifications/?ticket=<ticket> 

162 

163 티켓 발급: POST /api/v1/realtime/ws-ticket/ 

164 """ 

165 from urllib.parse import parse_qs 

166 

167 from django.contrib.auth import get_user_model 

168 from django.core.cache import cache 

169 

170 User = get_user_model() 

171 

172 query_string = self.scope.get("query_string", b"").decode() 

173 params = parse_qs(query_string) 

174 ticket = params.get("ticket", [""])[0] 

175 if not ticket: 

176 logger.warning("ws_auth_missing_ticket") 

177 return None 

178 

179 cache_key = f"ws_ticket:{ticket}" 

180 user_id = await cache.aget(cache_key) 

181 if user_id is None: 

182 logger.warning("ws_auth_invalid_ticket", ticket_prefix=ticket[:10]) 

183 return None 

184 

185 # 1회용 — 즉시 삭제 

186 await cache.adelete(cache_key) 

187 

188 try: 

189 return await User.objects.aget(pk=user_id, is_active=True) 

190 except User.DoesNotExist: 

191 logger.warning("ws_auth_user_not_found", user_id=user_id) 

192 return None
