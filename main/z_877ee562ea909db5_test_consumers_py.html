127 async def test_status_change(self): 

128 """pending → in_progress → done 상태 변경 시 각 상태마다 status 이벤트를 전송합니다.""" 

129 initial_ujd = self._make_ujd_mock("pending", user_id=self.user.pk) 

130 ujd_in_progress = self._make_ujd_mock("in_progress") 

131 ujd_done = self._make_ujd_mock("done") 

132 consumer = self._build_consumer(self.user, uuid.uuid4()) 

133 

134 with ( 

135 patch.object( 

136 consumer, 

137 "_get_user_job_description", 

138 new=AsyncMock(side_effect=[initial_ujd, ujd_in_progress, ujd_done]), 

139 ), 

140 patch("api.v1.job_descriptions.consumers.asyncio.sleep", new=AsyncMock()), 

141 ): 

142 await consumer.stream() 

143 

144 # 초기 pending + in_progress + done = 총 3회 

145 self.assertEqual(consumer.send_event.call_count, 3) 

146 sent_statuses = [call[0][0]["collection_status"] for call in consumer.send_event.call_args_list] 

147 self.assertEqual(sent_statuses, ["pending", "in_progress", "done"]) 

148 sent_events = [call[1]["event"] for call in consumer.send_event.call_args_list] 

149 self.assertEqual(sent_events, ["status", "status", "status"])
