11class GrantDailySubscriptionTicketsService(BaseService): 

12 """활성 구독 정책에 따라 모든 해당 사용자에게 일일 티켓을 리셋 지급한다. 

13 

14 매일 KST 00:00에 Celery Beat 태스크에 의해 호출된다. 

15 

16 일일 티켓은 누적이 아닌 리셋 방식이다: 

17 - UserTicket.daily_count를 정책의 daily_ticket_amount로 덮어쓴다. 

18 - 구매 티켓(purchased_count)은 영향받지 않는다. 

19 """ 

20 

21 def execute(self): 

22 now = timezone.now() 

23 policies = list(SubscriptionPlanTicketPolicy.objects.filter(is_active=True)) 

24 

25 if not policies: 

26 return 0 

27 

28 # plan_type → daily_ticket_amount 매핑 

29 policy_map = {p.plan_type: p.daily_ticket_amount for p in policies} 

30 

31 # 모든 활성 정책의 plan_type을 한 번에 조회 (N+1 방지) 

32 active_subscriptions = list( 

33 Subscription.objects.filter( 

34 plan_type__in=policy_map.keys(), 

35 started_at__lte=now, 

36 ).filter(Q(expires_at__isnull=True) | Q(expires_at__gt=now)).select_related("user") 

37 ) 

38 

39 granted_count = 0 

40 for subscription in active_subscriptions: 

41 daily_amount = policy_map.get(subscription.plan_type, 0) 

42 if daily_amount <= 0: 

43 continue 

44 

45 UserTicket.objects.update_or_create( 

46 user=subscription.user, 

47 defaults={"daily_count": daily_amount}, 

48 create_defaults={ 

49 "daily_count": daily_amount, 

50 "purchased_count": 0 

51 }, 

52 ) 

53 

54 granted_count += 1 

55 logger.info( 

56 "Daily ticket reset | user=%s plan=%s amount=%d", 

57 subscription.user_id, 

58 subscription.plan_type, 

59 daily_amount, 

60 ) 

61 

62 return granted_count
