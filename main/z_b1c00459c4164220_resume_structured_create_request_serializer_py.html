1"""structured 모드 이력서 생성 요청 serializer.

3사용자가 이미 구조화 입력 UI 로 직접 작성한 정규화 데이터를 한 번에 수신한다. 

4LLM 분석 단계는 생략되고, 저장 직후 곧바로 임베딩 파이프라인이 발행된다.

103 def create(self, validated_data): 

104 # 1) canonical N:M 은 nested 가 아니라 별도 서비스로 처리하기 위해 pop 

105 skills = validated_data.pop("skills", None) 

106 industry_domains = validated_data.pop("industry_domains", None) 

107 keywords = validated_data.pop("keywords", None) 

108 job_category_name = validated_data.pop("resume_job_category_name", "") or "" 

109 

110 # 2) structured 전용 기본 상태 세팅 

111 validated_data["type"] = ResumeType.STRUCTURED 

112 validated_data["source_mode"] = ResumeSourceMode.STRUCTURED 

113 validated_data["analysis_status"] = AnalysisStatus.COMPLETED 

114 validated_data["analysis_step"] = AnalysisStep.DONE 

115 validated_data["is_parsed"] = True 

116 validated_data["analyzed_at"] = timezone.now() 

117 

118 # 3) drf-writable-nested 의 create() 가 nested 자식 저장까지 모두 처리 

119 resume = super().create(validated_data) 

120 

121 # 4) canonical 참조 테이블 경유 N:M 은 기존 replace 서비스로 위임 

122 if skills: 

123 ReplaceResumeSkillsService(resume=resume, skills=skills).perform() 

124 if industry_domains: 

125 ReplaceResumeIndustryDomainsService(resume=resume, industry_domains=industry_domains).perform() 

126 if keywords: 

127 ReplaceResumeKeywordsService(resume=resume, keywords=keywords).perform() 

128 

129 # 5) 이름 기반 job category upsert (chip/autocomplete 으로 선택된 라벨) 

130 # _id 로 직접 붙이지 못한 경우에만 name lookup-or-create 를 태운다. 

131 if job_category_name and not resume.resume_job_category_id: 

132 UpdateResumeJobCategoryService(resume=resume, name=job_category_name).perform() 

133 resume.refresh_from_db() 

134 

135 return resume
