1""" 

2오래된 녹화 레코드를 ABANDONED 상태로 전환하는 Celery 태스크. 

3 

4INITIATED 또는 UPLOADING 상태로 1시간 이상 머물러 있는 녹화를 

5ABANDONED 으로 일괄 업데이트한다. 30분마다 실행된다. 

6""" 

7 

8from datetime import timedelta 

9 

10from celery.schedules import crontab 

11from common.tasks.base_scheduled_task import BaseScheduledTask 

12from config.celery import app 

13from django.utils import timezone 

14from interviews.enums import RecordingStatus 

15from interviews.models import InterviewRecording 

16 

17STALE_THRESHOLD = timedelta(hours=1) 

18 

19 

20class CleanupStaleRecordingsTask(BaseScheduledTask): 

21 """1시간 이상 INITIATED/UPLOADING 상태인 녹화를 ABANDONED 으로 전환한다. 30분마다 실행.""" 

22 

23 schedule = crontab(minute="*/30") 

24 

25 def run(self): 

26 cutoff = timezone.now() - STALE_THRESHOLD 

27 stale_recordings = InterviewRecording.objects.filter( 

28 status__in=[RecordingStatus.INITIATED, RecordingStatus.UPLOADING], 

29 created_at__lt=cutoff, 

30 ) 

31 count = stale_recordings.update(status=RecordingStatus.ABANDONED) 

32 return {"abandoned_count": count} 

33 

34 

35RegisteredCleanupStaleRecordingsTask = app.register_task(CleanupStaleRecordingsTask())
