10class GetCurrentSubscriptionService(BaseService): 

11 """현재 사용자의 활성 구독을 반환한다. 

12 

13 유료 구독이 활성 중이면 유료 구독을 우선 반환하고, 

14 없으면 무료 구독을 반환한다. 

15 활성 여부는 날짜(started_at, expires_at)로 판단하며 DB의 status 컬럼은 없다.

17 

18 def execute(self):

20 

21 # 현재 시각 기준 유효한 구독: 시작됐고 아직 만료되지 않은 것 

22 active_qs = Subscription.objects.filter( 

23 user=self.user, 

24 started_at__lte=now, 

25 ).filter(Q(expires_at__isnull=True) | Q(expires_at__gt=now)) 

26 

27 # 유료 구독 우선 

28 paid_subscription = active_qs.exclude(plan_type=PlanType.FREE).order_by("-started_at").first() 

29 if paid_subscription: 

30 return paid_subscription 

31 

32 free_subscription = active_qs.filter(plan_type=PlanType.FREE).order_by("-started_at").first() 

33 if free_subscription: 

34 return free_subscription 

35 

36 return CreateFreeSubscriptionService(user=self.user).perform()
