1"""ChunkPoolBuilder 서비스. 

2 

3ResumeEmbedding 테이블과 JobDescription 모델에서 청크를 수집하여 

4청크 풀을 구성한다. TEXT, SUMMARY, BASIC_INFO, CERTIFICATION 유형은 제외한다 

5""" 

6 

7from __future__ import annotations 

8 

9import logging 

10 

11from interviews.schemas.chunk_item import ChunkItem 

12from resumes.enums import ChunkType 

13from resumes.models import ResumeEmbedding 

14 

15logger = logging.getLogger(__name__) 

16 

17_EXCLUDED_CHUNK_TYPES = {ChunkType.TEXT, ChunkType.SUMMARY, ChunkType.BASIC_INFO, ChunkType.CERTIFICATION} 

18 

19_JD_FIELD_LABELS: dict[str, str] = { 

20 "duties": "담당업무", 

21 "requirements": "자격요건", 

22 "preferred": "우대사항", 

23} 

24 

25 

26class ChunkPoolBuilder: 

27 """이력서와 채용공고에서 청크를 수집하여 청크 풀을 구성한다.""" 

28 

29 def build(self, resume, job_description) -> list[ChunkItem]: 

30 """이력서 청크와 채용공고 청크를 합쳐 하나의 청크 풀을 반환한다.""" 

31 resume_chunks = self._build_resume_chunks(resume) 

32 jd_chunks = self._build_jd_chunks(job_description) 

33 return resume_chunks + jd_chunks 

34 

35 def _build_resume_chunks(self, resume) -> list[ChunkItem]: 

36 """ResumeEmbedding에서 TEXT/SUMMARY/BASIC_INFO/CERTIFICATION을 제외한 청크를 조회한다.""" 

37 if resume is None: 

38 return [] 

39 try: 

40 embeddings = ResumeEmbedding.objects.filter( 

41 resume=resume 

42 ).exclude(chunk_type__in=[ct.value for ct in _EXCLUDED_CHUNK_TYPES], ) 

43 return [ 

44 ChunkItem( 

45 source_label="이력서", 

46 type_label=ChunkType(emb.chunk_type).label, 

47 text=emb.context.strip(), 

48 ) for emb in embeddings if emb.context.strip() 

49 ] 

50 except Exception: 

51 logger.exception("이력서 청크 조회 실패") 

52 return [] 

53 

54 def _build_jd_chunks(self, job_description) -> list[ChunkItem]: 

55 """JobDescription의 duties/requirements/preferred에서 비어있지 않은 필드만 청크로 생성한다.""" 

56 if job_description is None: 

57 return [] 

58 try: 

59 chunks: list[ChunkItem] = [] 

60 for field_name, label in _JD_FIELD_LABELS.items(): 

61 value = getattr(job_description, field_name, "") 

62 if value and value.strip(): 

63 chunks.append(ChunkItem( 

64 source_label="채용공고", 

65 type_label=label, 

66 text=value.strip(), 

67 )) 

68 return chunks 

69 except Exception: 

70 logger.exception("채용공고 청크 조회 실패") 

71 return []
