1"""꼬리질문 생성기. 

2 

3직전 질문·답변과 대화 이력을 분석하여 아직 검증되지 않은 영역을 집중적으로 파고드는 

4꼬리질문을 Structured Output으로 생성한다. 

5""" 

6 

7from __future__ import annotations 

8 

9import logging 

10 

11from common.llm_client import get_llm 

12from interviews.schemas.followup_generator_input import FollowUpGeneratorInput 

13from interviews.schemas.followup_generator_output import FollowUpGeneratorOutput 

14from interviews.schemas.followup_question import FollowUpQuestion 

15from interviews.services.llm.prompt_registry import PromptRegistry 

16from interviews.services.llm.token_tracker import TokenUsageCallback 

17from langchain_core.messages import HumanMessage, SystemMessage 

18from pydantic import BaseModel 

19 

20logger = logging.getLogger(__name__) 

21 

22# ── Structured Output 스키마 ───────────────────────────────────────────────── 

23 

24 

25class _FollowUpItem(BaseModel): 

26 question: str 

27 rationale: str 

28 

29 

30class _FollowUpOutputSchema(BaseModel): 

31 followup_questions: list[_FollowUpItem] 

32 

33 

34# ── 난이도별 프롬프트 ───────────────────────────────────────────────────────── 

35 

36_registry = PromptRegistry() 

37 

38 

39class FollowUpQuestionGenerator: 

40 """직전 답변 기반 꼬리질문 생성기.""" 

41 

42 def generate( 

43 self, 

44 input_data: FollowUpGeneratorInput, 

45 callback: TokenUsageCallback | None = None, 

46 ) -> FollowUpGeneratorOutput: 

47 llm = get_llm() 

48 if callback: 

49 llm = llm.with_config(callbacks=[callback]) 

50 

51 structured_llm = llm.with_structured_output(_FollowUpOutputSchema) 

52 messages = self._build_messages(input_data) 

53 

54 try: 

55 result: _FollowUpOutputSchema = structured_llm.invoke(messages) 

56 except Exception as e: 

57 raise RuntimeError(f"LLM 호출 실패: {e}") from e 

58 

59 questions = [ 

60 FollowUpQuestion(question=item.question, rationale=item.rationale) for item in result.followup_questions 

61 ] 

62 return FollowUpGeneratorOutput( 

63 followup_questions=questions, 

64 original_question=input_data.original_question, 

65 user_answer=input_data.user_answer, 

66 ) 

67 

68 def _build_messages(self, input_data: FollowUpGeneratorInput) -> list[SystemMessage | HumanMessage]: 

69 """system/user 역할을 분리한 메시지 리스트를 생성한다.""" 

70 system_prompt = _registry.get_followup_prompt(input_data.difficulty_level) 

71 

72 # User message: 참고 자료 + 대화 이력 + 지시사항 

73 parts: list[str] = [] 

74 

75 # 이력서·채용공고 참조 

76 parts.append( 

77 "## 참고 자료\n" 

78 f"### 이력서\n{input_data.resume_content}\n\n" 

79 f"### 채용공고(JSON)\n{input_data.job_description_content}\n\n" 

80 ) 

81 

82 # 대화 이력 (이전 Q&A) 

83 if input_data.history: 

84 parts.append("## 지금까지의 면접 대화\n") 

85 for i, h in enumerate(input_data.history): 

86 parts.append(f"Q{i + 1}: {h.get('question', '')}\n") 

87 if h.get("answer"): 

88 parts.append(f"A{i + 1}: {h.get('answer', '')}\n") 

89 parts.append("\n") 

90 

91 # 현재 질문·답변 

92 parts.append("## 현재 질문과 답변\n" 

93 f"질문: {input_data.original_question}\n" 

94 f"답변: {input_data.user_answer}\n\n") 

95 

96 # 꼬리질문 생성 지시 

97 parts.append( 

98 f"위 답변을 바탕으로 꼬리질문 {input_data.max_followup_questions_count}개를 생성하세요.\n" 

99 "각 꼬리질문에는 이 질문을 선택한 이유(rationale)를 함께 작성하세요.\n" 

100 "이미 다룬 주제를 반복하지 말고, 아직 검증되지 않은 영역에 집중하세요." 

101 ) 

102 

103 return [ 

104 SystemMessage(content=system_prompt), 

105 HumanMessage(content="".join(parts)), 

106 ]
