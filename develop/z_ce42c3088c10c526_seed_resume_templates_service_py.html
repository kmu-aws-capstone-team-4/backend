220class SeedResumeTemplatesService: 

221 """모든 Job에 대해 최소 1개의 템플릿을 보장한다.""" 

222 

223 @classmethod 

224 @transaction.atomic 

225 def perform(cls) -> int: 

226 created = 0 

227 for job in Job.objects.all(): 

228 if ResumeTextContentTemplate.objects.filter(job=job).exists(): 

229 continue 

230 

231 content = cls._find_content_for_job(job.name) 

232 ResumeTextContentTemplate.objects.create( 

233 job=job, 

234 title=job.name, 

235 content=content, 

236 display_order=0, 

237 ) 

238 created += 1 

239 return created 

240 

241 @staticmethod 

242 def _find_content_for_job(job_name: str) -> str: 

243 for keyword, content in TEMPLATE_CONTENT_BY_KEYWORD.items(): 

244 if keyword in job_name: 

245 return content 

246 return DEFAULT_TEMPLATE_CONTENT
