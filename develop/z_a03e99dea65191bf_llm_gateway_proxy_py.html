1"""LiteLLM Gateway 어드민 UI 프록시 — Django admin 인증(is_staff) 전용. 

2 

3Flower 프록시와 다른 점: 

4- LiteLLM 은 FastAPI ``root_path`` 메커니즘을 사용하므로 라우트는 여전히 ``/ui`` 등 

5 원래 경로에 정의되어 있다. 따라서 프록시는 ``/admin/llm-gateway/`` prefix 를 

6 **벗기고** 내부 LiteLLM 으로 forward 해야 한다. 

7- LiteLLM UI 는 ``token`` 쿠키(JWT) 기반 세션을 사용하므로 응답의 ``Set-Cookie`` 를 

8 반드시 그대로 전달해야 한다. 

9"""

25# 빈 문자열은 dev/test 환경에서 LiteLLM 이 미배포 상태임을 의미. 검증 skip. 

26# 비어 있지 않으면 import 시점에 scheme 검증 — 잘못된 설정은 즉시 실패.

28 _scheme = urlparse(_LLM_GATEWAY_BASE).scheme 

29 if _scheme not in {"http", "https"}: 

30 raise ValueError(f"LLM_GATEWAY_INTERNAL_URL scheme must be http or https, got: {_scheme!r}")

32# 프록시 시 제거할 hop-by-hop / 충돌 헤더 (요청) 

33_REQUEST_SKIP_HEADERS = frozenset({ 

34 "host", 

35 "content-length", 

36 "accept-encoding", 

37 "connection", 

38 "transfer-encoding",

41# 응답에서 그대로 forward 하면 안 되는 헤더. 

42# ``set-cookie`` 는 SimpleCookie 로 별도 처리 (다중값 보존). 

43# ``content-encoding`` / ``content-length`` 는 urllib 가 이미 압축 해제한 본문을 

44# 다시 보내므로 빼야 한다. 

45_RESPONSE_SKIP_HEADERS = frozenset( 

46 { 

47 "connection", 

48 "transfer-encoding", 

49 "content-encoding", 

50 "content-length", 

51 "keep-alive", 

52 "set-cookie",

57@csrf_exempt 

58def llm_gateway_proxy(request: HttpRequest, path: str = "") -> HttpResponse: 

59 """내부 LiteLLM Gateway 서비스로 요청을 포워딩합니다. 

60 

61 ``/admin/llm-gateway/<path>`` 로 들어온 요청에서 prefix 를 벗기고 

62 LiteLLM 내부 서비스(``LLM_GATEWAY_INTERNAL_URL``)의 ``/<path>`` 로 forward 한다. 

63 """ 

64 if not _LLM_GATEWAY_BASE: 

65 return HttpResponse( 

66 content=b"LLM Gateway is not configured in this environment.", 

67 content_type="text/plain", 

68 status=503, 

69 ) 

70 

71 target = f"{_LLM_GATEWAY_BASE}/{path}" 

72 if qs := request.META.get("QUERY_STRING"): 

73 target += f"?{qs}" 

74 

75 headers = {k: v for k, v in request.headers.items() if k.lower() not in _REQUEST_SKIP_HEADERS} 

76 

77 # POST/PUT 등 body 가 있는 요청은 data 를 반드시 bytes 로 전달해야 

78 # urllib 가 메서드를 GET 으로 바꾸지 않는다. 

79 body = request.body if request.method in {"POST", "PUT", "PATCH", "DELETE"} else None 

80 

81 try: 

82 proxy_req = urllib.request.Request( 

83 url=target, 

84 method=request.method, 

85 data=body, 

86 headers=headers, 

87 ) 

88 with urllib.request.urlopen(proxy_req, timeout=30) as resp: # nosec B310 

89 return _build_proxy_response(resp.read(), resp.status, resp.headers) 

90 except urllib.error.HTTPError as e: 

91 body_content = e.read() if hasattr(e, "read") else b"" 

92 return _build_proxy_response(body_content, e.code, e.headers) 

93 except Exception: 

94 logger.exception( 

95 "llm_gateway_proxy_failed method=%s target=%s", 

96 request.method, 

97 target, 

98 ) 

99 return HttpResponse(status=502)

102def _build_proxy_response(content: bytes, status: int, upstream_headers) -> HttpResponse: 

103 """업스트림 응답을 Django HttpResponse 로 빌드 — Set-Cookie 다중값 정확히 전달.""" 

104 content_type = "text/html" 

105 if upstream_headers is not None: 

106 content_type = upstream_headers.get("Content-Type", "text/html") 

107 

108 response = HttpResponse(content=content, content_type=content_type, status=status) 

109 

110 if upstream_headers is None: 

111 return response 

112 

113 # urllib 의 HTTPMessage 는 동일 헤더가 여러 번 등장하면 .items() / .keys() 가 

114 # 각 인스턴스를 별도 항목으로 yield 한다. 단순 ``response[key] = value`` 루프는 

115 # 마지막 값으로 덮어쓰므로 ``Vary``, ``Cache-Control`` 같은 다중값을 잃는다. 

116 # RFC 7230 §3.2.2 에 따라 combinable 헤더는 ", " 로 join 해 한 번만 set 한다. 

117 # ``Set-Cookie`` 는 join 불가능 — 아래 SimpleCookie 블록에서 별도 처리. 

118 seen_keys: set[str] = set() 

119 for key in upstream_headers.keys(): 

120 lowered = key.lower() 

121 if lowered in _RESPONSE_SKIP_HEADERS or lowered in seen_keys: 

122 continue 

123 seen_keys.add(lowered) 

124 if hasattr(upstream_headers, "get_all"): 

125 values = upstream_headers.get_all(key) or [] 

126 else: 

127 single = upstream_headers.get(key) 

128 values = [single] if single is not None else [] 

129 values = [v for v in values if v is not None] 

130 if values: 

131 response[key] = ", ".join(values) 

132 

133 # LiteLLM 의 ``token`` / ``litellm_cp_return_to`` 쿠키 등을 SimpleCookie 로 

134 # 파싱해 morsel 속성(HttpOnly, Secure, SameSite ...) 을 보존하면서 재설정. 

135 if hasattr(upstream_headers, "get_all"): 

136 set_cookie_values = upstream_headers.get_all("Set-Cookie") or [] 

137 else: 

138 single_value = upstream_headers.get("Set-Cookie") 

139 set_cookie_values = [single_value] if single_value else [] 

140 

141 for cookie_header in set_cookie_values: 

142 cookie = SimpleCookie() 

143 cookie.load(cookie_header) 

144 for name, morsel in cookie.items(): 

145 response.cookies[name] = morsel.value 

146 for attr_key, attr_val in morsel.items(): 

147 if attr_val != "": 

148 response.cookies[name][attr_key] = attr_val 

149 

150 return response
