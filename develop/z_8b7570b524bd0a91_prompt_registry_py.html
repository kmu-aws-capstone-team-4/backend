111 def get_question_prompt(self, difficulty_level: str, company_name: str = "", job_title: str = "") -> str: 

112 if difficulty_level not in self.VALID_LEVELS: 

113 raise ValueError(f"지원하지 않는 난이도: {difficulty_level}. 지원 목록: {', '.join(self.VALID_LEVELS)}") 

114 prompt = self._QUESTION_PROMPTS[difficulty_level] 

115 return prompt.format(company_name=company_name or "회사", job_title=job_title or "채용 포지션") 

116 

117 def get_followup_prompt(self, difficulty_level: str) -> str: 

118 if difficulty_level not in self.VALID_LEVELS: 

119 raise ValueError(f"지원하지 않는 난이도: {difficulty_level}. 지원 목록: {', '.join(self.VALID_LEVELS)}") 

120 return self._FOLLOWUP_PROMPTS[difficulty_level]

123 return self._FULL_PROCESS_PROMPT
