66class FollowUpQuestionGenerator: 

67 """직전 답변 기반 꼬리질문 생성기.""" 

68 

69 def generate( 

70 self, 

71 input_data: FollowUpGeneratorInput, 

72 callback: TokenUsageCallback | None = None, 

73 ) -> FollowUpGeneratorOutput: 

74 llm = get_llm() 

75 if callback: 

76 llm = llm.with_config(callbacks=[callback]) 

77 

78 structured_llm = llm.with_structured_output(_FollowUpOutputSchema) 

79 prompt = self._build_prompt(input_data) 

80 

81 try: 

82 result: _FollowUpOutputSchema = structured_llm.invoke(prompt) 

83 except Exception as e: 

84 raise RuntimeError(f"LLM 호출 실패: {e}") from e 

85 

86 questions = [ 

87 FollowUpQuestion(question=item.question, rationale=item.rationale) for item in result.followup_questions 

88 ] 

89 return FollowUpGeneratorOutput( 

90 followup_questions=questions, 

91 original_question=input_data.original_question, 

92 user_answer=input_data.user_answer, 

93 ) 

94 

95 def _build_prompt(self, input_data: FollowUpGeneratorInput) -> str: 

96 system_prompt = _SYSTEM_PROMPTS.get(input_data.difficulty_level, _SYSTEM_PROMPTS["normal"]) 

97 parts: list[str] = [system_prompt, "\n"] 

98 

99 # 이력서·채용공고 참조 

100 parts.append( 

101 "## 참고 자료\n" 

102 f"### 이력서\n{input_data.resume_content}\n\n" 

103 f"### 채용공고(JSON)\n{input_data.job_description_content}\n\n" 

104 ) 

105 

106 # 대화 이력 (이전 Q&A) 

107 if input_data.history: 

108 parts.append("## 지금까지의 면접 대화\n") 

109 for i, h in enumerate(input_data.history): 

110 parts.append(f"Q{i + 1}: {h.get('question', '')}\n") 

111 if h.get("answer"): 

112 parts.append(f"A{i + 1}: {h.get('answer', '')}\n") 

113 parts.append("\n") 

114 

115 # 현재 질문·답변 

116 parts.append("## 현재 질문과 답변\n" 

117 f"질문: {input_data.original_question}\n" 

118 f"답변: {input_data.user_answer}\n\n") 

119 

120 # 꼬리질문 생성 지시 

121 parts.append( 

122 f"위 답변을 바탕으로 꼬리질문 {input_data.max_followup_questions_count}개를 생성하세요.\n" 

123 "각 꼬리질문에는 이 질문을 선택한 이유(rationale)를 함께 작성하세요.\n" 

124 "이미 다룬 주제를 반복하지 말고, 아직 검증되지 않은 영역에 집중하세요." 

125 ) 

126 

127 return "".join(parts)
