113 def mark_paused(self, reason: str) -> None: 

114 """면접 세션을 일시정지 상태로 변경한다.""" 

115 if self.interview_session_status != InterviewSessionStatus.IN_PROGRESS: 

116 raise ValueError("진행 중인 세션만 일시정지할 수 있습니다.") 

117 

118 self.interview_session_status = InterviewSessionStatus.PAUSED 

119 self.paused_at = timezone.now() 

120 self.pause_count += 1 

121 self.pause_reason = reason 

122 self.save(update_fields=["interview_session_status", "paused_at", "pause_count", "pause_reason", "updated_at"]) 

123 

124 def mark_resumed(self) -> None: 

125 """면접 세션을 다시 진행 상태로 변경한다.""" 

126 if self.interview_session_status != InterviewSessionStatus.PAUSED or self.paused_at is None: 

127 raise ValueError("일시정지 상태가 아닙니다.") 

128 

129 now = timezone.now() 

130 duration_ms = int((now - self.paused_at).total_seconds() * 1000) 

131 self.total_paused_duration_ms += duration_ms 

132 

133 self.interview_session_status = InterviewSessionStatus.IN_PROGRESS 

134 self.paused_at = None 

135 self.pause_reason = "" 

136 self.save( 

137 update_fields=["interview_session_status", "paused_at", "pause_reason", "total_paused_duration_ms", "updated_at"] 

138 )
