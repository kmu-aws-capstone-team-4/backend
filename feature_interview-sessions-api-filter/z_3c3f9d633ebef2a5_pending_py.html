1""" 

2N+1 감지 후 요청 완료 시점까지 알림을 지연시키는 pending 저장소. 

3 

4emit() 시점에는 N+1 반복 쿼리가 아직 실행 중이므로 SQL 을 바로 수집할 수 없다. 

5request_finished signal 에서 flush() 를 호출해 SQL 을 수집하고 Celery task 를 전송한다. 

6""" 

7 

8from __future__ import annotations 

9 

10import time 

11from dataclasses import dataclass 

12 

13from .request import get_request_id 

14from .sql import collect_sql 

15 

16_TTL = 3600 

17_CLEANUP_INTERVAL = 100 

18 

19 

20@dataclass 

21class PendingAlert: 

22 model: str 

23 field: str 

24 path: str 

25 method: str 

26 stacktrace: str 

27 query_snapshot_index: int 

28 registered_at: float = 0.0 

29 

30 def __post_init__(self) -> None: 

31 if not self.registered_at: 

32 self.registered_at = time.monotonic() 

33 

34 

35# 모듈 레벨 상태 — 프로세스 수명 동안 유지 

36_emit_count = 0 

37_seen: dict[tuple[str, str, str], float] = {} 

38_pending: dict[str, list[PendingAlert]] = {} 

39 

40 

41def is_duplicate(request_id: str, model: str, field: str) -> bool: 

42 """같은 요청에서 동일한 (model, field) 조합이 이미 감지됐으면 True.""" 

43 global _emit_count 

44 

45 key = (request_id, model, field) 

46 if key in _seen: 

47 return True 

48 

49 _seen[key] = time.monotonic() 

50 _emit_count += 1 

51 if _emit_count % _CLEANUP_INTERVAL == 0: 

52 _evict_expired() 

53 return False 

54 

55 

56def register(request_id: str, alert: PendingAlert) -> None: 

57 """pending 저장소에 alert 를 등록한다.""" 

58 _pending.setdefault(request_id, []).append(alert) 

59 

60 

61def flush(request_id: str) -> None: 

62 """요청 완료 시 pending alert 를 처리하고 Celery task 를 전송한다.""" 

63 alerts = _pending.pop(request_id, []) 

64 if not alerts: 

65 return 

66 

67 from common.tasks.send_nplusone_alert_task import RegisteredSendNPlusOneAlertTask 

68 

69 for alert in alerts: 

70 RegisteredSendNPlusOneAlertTask.delay( 

71 model=alert.model, 

72 field=alert.field, 

73 path=alert.path, 

74 method=alert.method, 

75 stacktrace=alert.stacktrace, 

76 sql_log=collect_sql(alert.model, alert.field, alert.query_snapshot_index), 

77 )

80def connect_signal() -> None: 

81 """request_finished Django signal 에 flush 를 연결한다. 최초 1회만 호출한다.""" 

82 from django.core.signals import request_finished 

83 

84 def on_request_finished(sender, **kwargs): # noqa: ANN001 

85 if request_id := get_request_id(): 

86 flush(request_id) 

87 

88 request_finished.connect(on_request_finished, weak=False)

91def _evict_expired() -> None: 

92 """TTL 이 지난 _seen / _pending 항목을 제거한다.""" 

93 cutoff = time.monotonic() - _TTL 

94 

95 # _seen 정리 

96 expired_seen = [k for k, ts in _seen.items() if ts < cutoff] 

97 for k in expired_seen: 

98 del _seen[k] 

99 

100 # _pending 정리 — request_finished 가 호출되지 않아 남은 항목 제거 

101 expired_pending = [rid for rid, alerts in _pending.items() if alerts and alerts[0].registered_at < cutoff] 

102 for rid in expired_pending: 

103 del _pending[rid]
